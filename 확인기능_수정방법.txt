def check_rows(table, column, ids, checker):
    """
    ids 행 중 아직 미확인(column 이 NULL)인 행에만 확인자(checker)를 적습니다.
    이미 다른 사람이 확인한 행은 그대로 둡니다. 실제로 확인한 건수를 돌려줍니다.
    table, column 은 프로그램 안에서 정해진 값만 넣습니다. (화면에서 받은 값 금지)
    """
    conn = get_db()
    done = 0
    # SQLite 는 ? 를 한 번에 999개까지만 받아서 500개씩 나눠서 처리합니다.
    for i in range(0, len(ids), 500):
        part = ids[i:i + 500]
        marks = ','.join('?' * len(part))          # '?,?,?'
        cur = conn.execute('UPDATE {0} SET {1}=? WHERE {1} IS NULL AND id IN ({2})'
                           .format(table, column, marks), [checker] + part)
        done += cur.rowcount
    conn.commit()
    return done

# 분석현황 확인 칸: 주소의 kind → DB 칸 이름 (이 두 개 말고는 404)
CHECK_COLUMNS = {'start': 'start_checked_by', 'end': 'end_checked_by'}
CHECK_NAMES = {'start': '분석중', 'end': '분석종료'}


def check_column(kind):
    if kind not in CHECK_COLUMNS:
        abort(404)
    return CHECK_COLUMNS[kind]


@app.route('/analysis/<int:item_id>/check/<kind>', methods=['POST'])
@login_required
def analysis_check(item_id, kind):
    """[확인] 버튼: kind='start' 시작확인 / 'end' 종료확인"""
    db.check_rows('analyses', check_column(kind), [item_id], checker())
    return go_back('analysis')


@app.route('/analysis/<int:item_id>/uncheck/<kind>', methods=['POST'])
@login_required
def analysis_uncheck(item_id, kind):
    """[확인취소] 버튼"""
    db.execute('UPDATE analyses SET {}=NULL WHERE id=?'.format(check_column(kind)), (item_id,))
    return go_back('analysis')


@app.route('/analysis/check-all/<kind>', methods=['POST'])

======================================================================
 [직접 확인할 것] - 회사 PC에서 고친 뒤 눌러 볼 순서
======================================================================
계정 2개(A, B)로 확인합니다.

[3팀 신호처리]
  □ 3건 추가 → 카드 머리에 '미확인 3건', 행 배경이 연한 노랑
  □ 1건 [확인] → '✔ (내 급수 이름)' 표시, '미확인 2건'
  □ [취소] → 다시 미확인
  □ A가 화면을 열어 둔 채 B가 1건 추가 → A가 [미확인 전체 확인]
    → B가 추가한 건은 미확인으로 남음 (새로고침하면 보임)
  □ 확인된 건 [수정] → 저장 → 미확인으로 바뀜
  □ 어제 날짜에 미확인 건을 남기고 오늘로 이동 → 위에 '다른 날짜 미확인' 알림,
    날짜를 누르면 그 날로 이동
  □ 모두 확인하면 '모두 확인됨' 표시, 버튼 사라짐

[5팀 분석현황]
  □ A 담당 2건, B 담당 3건 등록 → A로 [내 분석] 보기에서
    버튼이 '미확인 2건 확인 (내 분석)', 옆에 '전체 보기에는 미확인 5건 →'
  □ 버튼을 누르면 확인창에 '[보기: 내 분석]', 처리 후 '분석중 2건 확인했습니다.'
    [전체] 보기에서 B의 3건은 미확인 그대로
  □ [사람 선택]에서 B를 고르면 버튼 글자가 '(B의 급수 이름)'으로 바뀜
  □ [분석종료] → 분석종료 표에 종료 미확인으로 나옴 → [확인]
  □ [분석중으로 되돌리기] → 시작확인은 남고, 다시 종료하면 종료확인은 비어 있음
  □ 확인된 건 [수정] → 저장 → 시작확인·종료확인 모두 비어 있음

[홈 화면]
  □ 신호처리·분석현황 카드 제목 옆에 미확인 배지, 모두 확인하면 사라짐
  □ 엑셀 다운로드·인쇄는 이전과 똑같이 나옴


======================================================================
 [에러가 날 때 자주 있는 원인]
======================================================================
  - 서버가 안 켜짐 / SyntaxError, IndentationError
      → app.py, db.py 들여쓰기(공백 4칸) 확인. 탭 문자가 섞였는지 확인.
  - "sqlite3.OperationalError: no such column: checked_by" (또는 start_checked_by)
      → 옛 worklog.db 가 남아 있음. 서버 끄고 worklog.db 삭제 후 다시 켜기.
      → 그래도 나면 db.py SCHEMA 의 쉼표(,) 빠짐 확인 (1-1, 1-2).
  - "UndefinedError: 'check_cell' is undefined"
      → signal.html / analysis.html 맨 위 import 줄에 check_cell 빠짐 (4-1, 5-1).
  - "BuildError: Could not build url for endpoint 'signal_check'" 등
      → app.py 에 라우트 추가가 빠졌거나 함수 이름 오타 (2-5, 2-9).
  - 화면에 한글/✔ 가 깨짐
      → 템플릿 파일을 UTF-8 로 저장했는지 확인.
  - 에러 내용은 검은 창 마지막 줄, 또는 숨김 실행이면 server.log 에 있습니다.
